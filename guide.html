STEP 1

ติดตั้งแพ็กเกจ

พร้อมรัน

ต้องใช้ Node.js 18 ขึ้นไป (หรือ Bun, Deno, Cloudflare Workers, เบราว์เซอร์รุ่นใหม่) แพ็กเกจไม่มี runtime dependency และรองรับทั้ง ESM และ CommonJS

npm install lsupergen-sdk
# หรือ
pnpm add lsupergen-sdk
yarn add lsupergen-sdk
bun add lsupergen-sdk

ปุ่มรันจริงจะโหลดไฟล์ dist/index.js ของ lsupergen-sdk@0.1.0 (ไฟล์เดียวกับใน npm tarball) แล้วตรวจว่า export ครบ

STEP 2

สร้าง API key

พร้อมรัน

API key ผูกกับบัญชีที่เข้าสู่ระบบอยู่ มีอายุ 30 วัน และแสดงเพียงครั้งเดียว หน้านี้เก็บ key ไว้ในหน่วยความจำของแท็บนี้เท่านั้น (ไม่บันทึกลงเบราว์เซอร์)

อย่า commit key ลง Git และอย่าฝัง key ในโค้ดฝั่งเบราว์เซอร์ที่เผยแพร่ ให้เก็บเป็น environment variable หรือ Cloudflare Secret

STEP 3

ตั้งค่า client

ตั้ง environment variable สองตัว แล้ว client จะอ่านค่าเองโดยอัตโนมัติ

LSUPERGEN_API_KEY=lsg_your_key_here
LSUPERGEN_BASE_URL={{BASE}}
import { Lsupergen } from 'lsupergen-sdk';

const client = new Lsupergen({
  apiKey: process.env.LSUPERGEN_API_KEY,   // ค่าเริ่มต้นอ่านจาก LSUPERGEN_API_KEY
  baseURL: process.env.LSUPERGEN_BASE_URL, // {{BASE}}
  timeout: 60_000,                         // ms
  maxRetries: 2,                           // retry 408/409/429/5xx อัตโนมัติ
});

เบราว์เซอร์ / Cloudflare Workers / Deno

lsupergen-sdk 0.1.0 เรียก fetch ที่เก็บไว้โดยไม่ผูก this ซึ่ง Node.js ยอมให้ แต่เบราว์เซอร์และ Cloudflare Workers จะปฏิเสธด้วย Illegal invocation (SDK จะโยน ConnectionError) ให้ส่ง fetch เข้าไปเองตามนี้ ปุ่มรันจริงในหน้านี้ก็ใช้วิธีเดียวกัน

const client = new Lsupergen({
  apiKey,
  baseURL: '{{BASE}}',
  fetch: (...args) => fetch(...args), // จำเป็นสำหรับเบราว์เซอร์และ Workers ในเวอร์ชัน 0.1.0
});

ต้องตั้ง baseURL เป็น … เสมอ เพราะค่าเริ่มต้นในแพ็กเกจ 0.1.0 ชี้ไปที่ https://api.lsupergen.com ซึ่งไม่ใช่ API ของเว็บไซต์นี้

STEP 4

ตรวจสถานะ API

พร้อมรัน

GET /health ไม่ต้องใช้ key ใช้ตรวจว่าเชื่อมต่อได้ และบริการ AI พร้อมหรือไม่ (ai_ready)

const health = await client.get('/health');
console.log(health.ok, health.ai_ready);
STEP 5

ยืนยันตัวตนด้วย key

พร้อมรัน

GET /me ส่ง Authorization: Bearer <key> ให้อัตโนมัติ และตอบกลับบัญชีเจ้าของ key พร้อมวันหมดอายุ

const me = await client.get('/me');
console.log(me.user.email, me.key.expires_at);
STEP 6

เรียก AI Chat

พร้อมรัน

POST /chat ส่งข้อความไปยังโมเดล AI จริงของระบบ ผลลัพธ์อยู่ใน message ถ้าบริการ AI ยังไม่พร้อม API จะตอบ 503 ตามจริง ไม่มีคำตอบปลอม

const reply = await client.post('/chat', {
  body: { message: 'สรุปจุดเด่นของ lsupergen-sdk ใน 1 ประโยค' },
});
console.log(reply.message);

// ต่อบทสนทนา: ส่ง messages สลับ user / assistant และจบด้วย user
await client.post('/chat', {
  body: {
    messages: [
      { role: 'user', content: 'ฉันชื่อแบงค์' },
      { role: 'assistant', content: 'ยินดีที่รู้จัก' },
      { role: 'user', content: 'ฉันชื่ออะไร' },
    ],
  },
});

// โหมดเครื่องมือ: tool = 'writer' | 'research' | 'url' | 'code'
await client.post('/chat', { body: { message: 'ค้นข่าว AI ล่าสุด', tool: 'research' } });
STEP 7

จัดการข้อผิดพลาด

พร้อมรัน

ทุก error สืบทอดจาก LsupergenError เมื่อเซิร์ฟเวอร์ตอบ non-2xx จะได้ APIError ที่มี status และ body ปุ่มรันจริงจะส่ง key ที่ไม่ถูกต้องเพื่อพิสูจน์ว่าได้ APIError 401

import { Lsupergen, APIError, TimeoutError, ConnectionError } from 'lsupergen-sdk';

try {
  await client.get('/me');
} catch (err) {
  if (err instanceof APIError) {
    console.error(err.status, err.body);   // เช่น 401 { error: 'invalid_api_key' }
  } else if (err instanceof TimeoutError) {
    // เกิน timeout
  } else if (err instanceof ConnectionError) {
    // เครือข่ายมีปัญหา
  }
}
STEP 8

ตรวจสอบทั้งคู่มือ

พร้อมรัน

รันขั้นที่ 1, 4, 5, 6 และ 7 ต่อกันด้วย SDK จริง แล้วสร้าง Proof report (JSON) ที่มีเวลา สถานะ HTTP และผลลัพธ์ของแต่ละขั้น ต้องมี API key จากขั้นที่ 2 ก่อน

STEP 9

ใช้ใน Node.js / Cloudflare Workers / curl

// quickstart.mjs — รัน: LSUPERGEN_API_KEY=lsg_... node quickstart.mjs
import { Lsupergen, APIError } from 'lsupergen-sdk';

const client = new Lsupergen({ baseURL: '{{BASE}}' });

try {
  const health = await client.get('/health');
  const me = await client.get('/me');
  const reply = await client.post('/chat', { body: { message: 'สวัสดี' } });
  console.log({ ai_ready: health.ai_ready, user: me.user.email, reply: reply.message });
} catch (err) {
  if (err instanceof APIError) console.error('API error', err.status, err.body);
  else throw err;
  process.exitCode = 1;
}
// quickstart.cjs
const { Lsupergen } = require('lsupergen-sdk');

const client = new Lsupergen({ baseURL: '{{BASE}}' });

client.post('/chat', { body: { message: 'สวัสดี' } })
  .then((reply) => console.log(reply.message))
  .catch((err) => { console.error(err.status, err.body); process.exitCode = 1; });
// src/index.js — เก็บ key ด้วย: npx wrangler secret put LSUPERGEN_API_KEY
import { Lsupergen, APIError } from 'lsupergen-sdk';

export default {
  async fetch(request, env) {
    const client = new Lsupergen({
      apiKey: env.LSUPERGEN_API_KEY,   // Workers ไม่มี process.env ต้องส่ง apiKey เอง
      baseURL: '{{BASE}}',
      fetch: (...args) => fetch(...args), // Workers ต้องใช้ fetch ที่ผูก this ถูกต้อง
    });
    const { message } = await request.json();
    try {
      const reply = await client.post('/chat', { body: { message } });
      return Response.json({ reply: reply.message });
    } catch (err) {
      const status = err instanceof APIError ? err.status : 502;
      return Response.json({ error: err.message }, { status });
    }
  },
};
curl -sS {{BASE}}/health

curl -sS {{BASE}}/me \
  -H "Authorization: Bearer $LSUPERGEN_API_KEY"

curl -sS {{BASE}}/chat \
  -H "Authorization: Bearer $LSUPERGEN_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"message":"สวัสดี"}'
STEP 10

API reference

Base URL: … ทุก response เป็น JSON และมี ok บอกผลลัพธ์

Method / PathAuthคำอธิบาย
GET /healthไม่ต้องสถานะบริการ: ok, ai_ready, package, time
GET /meBearerเจ้าของ key: user, key.expires_at
POST /chatBearerbody: message หรือ messages (สูงสุด 20 turn), tool, mode → { ok, status, message, output, sources }
POST /imageBearerbody: prompt → { ok, image: { mime_type, data_base64, filename } }

รหัสสถานะ

HTTPความหมาย
400 / 413ข้อมูลไม่ถูกต้อง หรือยาวเกินกำหนด (validation_error)
401ไม่มี key, key ผิด หรือหมดอายุ (invalid_api_key)
429เกินโควตา 10 คำขอ / 10 นาที ต่อ IP และเครื่องมือ นับในหน่วยความจำของ Worker แต่ละ isolate (ไม่ใช่โควตาต่อ API key หรือโควตารวม คนที่ใช้ IP เดียวกันจะนับรวมกัน) Retry-After อาจยาวถึง 600 วินาที ซึ่ง SDK ไม่รอให้ (รอเฉพาะ ≤ 60 วินาที) ต้องรอเองก่อนส่งใหม่
502 / 503บริการ AI ขัดข้อง หรือยังไม่ได้ตั้งค่า (service_unavailable)

ข้อควรรู้

  • API key เป็น token ที่เซ็นด้วยกุญแจของเซิร์ฟเวอร์ หมดอายุใน 30 วัน ถ้า key หลุดให้หยุดใช้และสร้างใหม่ ผู้ดูแลระบบยกเลิก key ทั้งหมดได้ด้วยการเปลี่ยน AUTH_SESSION_SECRET
  • ตั้ง maxRetries: 0 ถ้าไม่ต้องการ retry อัตโนมัติ และส่ง signal จาก AbortController เพื่อยกเลิกคำขอ
  • API เปิด CORS สำหรับการเรียกด้วย Bearer key แต่ควรเรียกจากฝั่งเซิร์ฟเวอร์เพื่อไม่ให้ key รั่ว
