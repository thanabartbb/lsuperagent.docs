การจัดการข้อผิดพลาด

ทุก error ของ SDK สืบทอดจาก LsupergenError

คลาสเกิดเมื่อข้อมูล
APIErrorเซิร์ฟเวอร์ตอบ non-2xxstatus, body, headers
TimeoutErrorเกิน timeoutmessage
ConnectionErrorเครือข่ายล้มเหลวcause คือ error ต้นทาง
LsupergenErrorตั้งค่าผิด เช่น ไม่มี apiKeymessage

ตัวอย่าง

import { APIError, TimeoutError, ConnectionError } from 'lsupergen-sdk';

try {
  await client.post('/chat', { body: { message: 'hi' } });
} catch (err) {
  if (err instanceof APIError) {
    if (err.status === 401) console.error('key ผิดหรือหมดอายุ');
    else if (err.status === 429) console.error('ส่งถี่เกินไป');
    else console.error(err.status, err.body);
  } else if (err instanceof TimeoutError) {
    console.error('หมดเวลา');
  } else if (err instanceof ConnectionError) {
    console.error('เชื่อมต่อไม่ได้', err.cause);
  }
}
