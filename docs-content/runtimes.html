import { Lsupergen, APIError } from 'lsupergen-sdk';

export default {
  async fetch(request, env) {
    const client = new Lsupergen({
      apiKey: env.LSUPERGEN_API_KEY, // Workers ไม่มี process.env
      baseURL: '{{BASE}}',
      fetch: (...args) => fetch(...args), // จำเป็นใน 0.1.0
    });
    const { message } = await request.json();
    try {
      const reply = await client.post('/chat', { body: { message } });
      return Response.json({ reply: reply.message });
    } catch (err) {
      return Response.json({ error: err.message }, { status: err instanceof APIError ? err.status : 502 });
    }
  },
};

เบราว์เซอร์

const client = new Lsupergen({
  apiKey,                 // อย่าฝัง key จริงในโค้ดที่เผยแพร่
  baseURL: '{{BASE}}',
  fetch: (...args) => fetch(...args), // จำเป็นใน 0.1.0
});

lsupergen-sdk 0.1.0: ในเบราว์เซอร์, Workers และ Deno ต้องส่ง fetch: (...args) => fetch(...args) ไม่งั้นทุกคำขอจะเป็น ConnectionError (Illegal invocation) เวอร์ชัน 0.1.1 ที่แก้บั๊กนี้ยังไม่ publish ขึ้น npm ระหว่างนี้ให้ใช้วิธีข้างบน
