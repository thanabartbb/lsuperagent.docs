POST /image

สร้างภาพจากคำอธิบาย ได้ไฟล์ PNG แบบ base64

Request body

promptstringคำอธิบายภาพ (≤ 12,000 ตัวอักษร)

ตัวอย่าง (Node.js)

import { writeFile } from 'node:fs/promises';

const result = await client.post('/image', {
  body: { prompt: 'แมวส้มนั่งบนโต๊ะทำงาน แสงยามเช้า' },
  timeout: 180_000,
});
await writeFile(result.image.filename, Buffer.from(result.image.data_base64, 'base64'));

Response 200

{
  "ok": true,
  "status": "completed",
  "image": { "mime_type": "image/png", "data_base64": "iVBORw0…", "filename": "lsuperagen-image.png", "revised_prompt": null }
}

การสร้างภาพใช้เวลานานกว่าแชท ควรตั้ง timeout อย่างน้อย 120–180 วินาที
