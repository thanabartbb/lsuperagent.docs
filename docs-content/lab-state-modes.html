Compaction

ฉบับร่างหนึ่งเขียนเงื่อนไข 80% ด้วยมือ อีกฉบับอ้าง /responses/compact ข้อยุติ: SDK มี OpenAIResponsesCompactionSession ที่เรียก responses.compact ให้เอง และรับฟังก์ชันตัดสินใจ should_trigger_compaction เราจึงใส่เงื่อนไข 80% เป็นฟังก์ชันนั้น ยืนยันแล้ว

# ---------------------------------------------------------------- Context budget ledger
def make_compaction_trigger(context_window_tokens: int, ratio: float = 0.8):
    """Build a should_trigger_compaction hook for OpenAIResponsesCompactionSession.

    The SDK passes a dict with `session_items` and `compaction_candidate_items`; it does not
    pass a token count, so this estimates ~4 characters per token.
    """

    def should_trigger(context: dict[str, Any]) -> bool:
        chars = sum(len(str(item)) for item in context["session_items"])
        return chars / 4 > ratio * context_window_tokens

    return should_trigger
from agents import OpenAIResponsesCompactionSession, SQLiteSession
from lab_patterns import make_compaction_trigger

session = OpenAIResponsesCompactionSession(
    "user-42",
    SQLiteSession("user-42"),
    should_trigger_compaction=make_compaction_trigger(context_window_tokens=128_000),
)
result = await Runner.run(agent, "hi", session=session)
